## Global Constraints

- **Task types:** `passage_questions`, `matching` (with "x" = `null` when `allowNone`) and `cloze` (gaps `typed` | `select` | `bank`, marked `{{n}}` in the text, numbered from 1).
- **Lesson pass rule:** ≥ 80% of items right is `correct`, ≥ 50% is `almost`, and anything else is `wrong`. Typed gaps use the fill-blank comparison.
- **The three types in lessons:** each is one review item, and all are test-out eligible. Phase 2 generation includes them, German-only.
- **Views:** never contain answers (`correctIndex`, `answers`, `answer`, `variants`).
- **Teil sets:**
  - usable in practice: `starter`, `approved`, `unreviewed`;
  - never in practice: `rejected` and `reserved`;
  - selection is unseen first. An exhausted part generates one German-only set into `unreviewed`, and if that fails, the least-recently-seen set comes back.
- **Runner timing:** always timed. Overtime is computed on the server from `started_at`. There's no pause, an unfinished attempt is never scored or shown, and attempts older than 3 h without `finished_at` are ignored.
- **Points:** a part's points are split evenly across its items.
- **Readiness:** the average percentage of the last 5 finished attempts per part, against 60%.
- **Existing rules:** `res.ok` and `role="alert"`; `delayedResponse`; all student text in en and de; admin in English; error codes through `errorBody`.

## Review Focus

1. **Double submit:** an attempt submitted twice is scored once. The second submit returns the stored review (Task 5).
2. **Submit after abandoning:** an answer submitted to an attempt that's older than 3 hours is rejected as `not_found`, and not scored (Task 5).
3. **Two gaps with the same bank word:** telc Sprachbausteine Teil 2 has 15 words for 10 gaps, and a word may be chosen for two gaps. Each gap is graded on its own, with no uniqueness enforced (Task 2).
4. **Empty answers:** an unanswered item counts as wrong, never as an error (Task 2).
5. **Removing a set type change:** an admin upload that changes a set's task type away from the part's type is rejected, with the set named (Task 9).

```ts
import { existsSync, readdirSync, readFileSync } from 'node:fs';
import { join } from 'node:path';
import type { CefrLevel, Track } from '../types';
import type { LocalizedText } from '../i18n/localizedText';

export type TaskTypeId = 'passage_questions' | 'matching' | 'cloze' | 'audio_questions' | 'letter' | 'spoken';
const TASK_TYPES: TaskTypeId[] = ['passage_questions', 'matching', 'cloze', 'audio_questions', 'letter', 'spoken'];

// Types that have a runner. Listening, Writing and Speaking add theirs.
export const RUNTIME_TASK_TYPES = new Set<TaskTypeId>(['passage_questions', 'matching', 'cloze']);

export interface ExamPart {
  id: string;
  skill: 'reading' | 'sprachbausteine' | 'listening' | 'writing' | 'speaking';
  title: LocalizedText;
  taskType: TaskTypeId;
  items: number;
  points: number;
  minutes: number;
  instructions: string;
  replays?: number;
}

export interface ExamBlock {
  id: 'written' | 'oral';
  title: LocalizedText;
  minutes: number;
  partIds: string[];
  passRatio: number;
}

export interface ExamFormat {
  id: string;
  track: Track;
  level: CefrLevel;
  name: string;
  sources: string[];
  blocks: ExamBlock[];
  parts: ExamPart[];
}

const DIR = () => join(process.cwd(), 'data', 'exam-formats');

export function formatId(track: Track, level: CefrLevel): string {
  return `${track}-${level.toLowerCase()}`;
}

export function readFormat(id: string): ExamFormat | null {
  const path = join(DIR(), `${id}.json`);
  return existsSync(path) ? (JSON.parse(readFileSync(path, 'utf8')) as ExamFormat) : null;
}

export function listFormats(): ExamFormat[] {
  return readdirSync(DIR())
    .filter((f) => f.endsWith('.json'))
    .sort()
    .map((f) => JSON.parse(readFileSync(join(DIR(), f), 'utf8')) as ExamFormat);
}

const positiveInt = (n: unknown) => typeof n === 'number' && Number.isInteger(n) && n > 0;

export function formatProblems(format: ExamFormat): string[] {
  const problems: string[] = [];
  const seen = new Set<string>();
  const byId = new Map<string, ExamPart>();
  for (const part of format.parts) {
    if (seen.has(part.id)) problems.push(`part ${part.id}: duplicate id`);
    seen.add(part.id);
    byId.set(part.id, part);
    if (!positiveInt(part.items)) problems.push(`part ${part.id}: items must be a positive whole number`);
    if (!(part.points > 0)) problems.push(`part ${part.id}: points must be positive`);
    if (!(part.minutes > 0)) problems.push(`part ${part.id}: minutes must be positive`);
    if (!TASK_TYPES.includes(part.taskType)) problems.push(`part ${part.id}: unknown task type ${part.taskType}`);
  }
  for (const block of format.blocks) {
    const missing = block.partIds.filter((id) => !byId.has(id));
    for (const id of missing) problems.push(`block ${block.id}: unknown part ${id}`);
    if (missing.length === 0) {
      const sum = block.partIds.reduce((s, id) => s + byId.get(id)!.minutes, 0);
      if (Math.abs(sum - block.minutes) > 1) problems.push(`block ${block.id}: minutes ${block.minutes} differ from its parts (${sum})`);
    }
  }
  return problems;
}
```

```ts
import type { LocalizedText } from '../i18n/localizedText';

export interface PassageQuestionsContent {
  instruction?: LocalizedText;
  passage: string;
  questions: { question: string; options: string[]; correctIndex: number }[];
}
export type PassageQuestionsView = Omit<PassageQuestionsContent, 'questions'> & { questions: { question: string; options: string[] }[] };
export interface PassageQuestionsAnswer {
  type: 'passage_questions';
  selected: (number | null)[];
}

const text = (v: unknown) => typeof v === 'string' && v.trim().length > 0;

export function validatePassageQuestions(content: unknown): string[] {
  const c = content as Partial<PassageQuestionsContent>;
  const problems: string[] = [];
  if (!text(c?.passage)) problems.push('passage must be a non-empty string');
  if (!Array.isArray(c?.questions) || c!.questions.length === 0) return [...problems, 'questions must be a non-empty list'];
  c!.questions.forEach((q, i) => {
    if (!text(q?.question)) problems.push(`question ${i + 1} needs text`);
    if (!Array.isArray(q?.options) || q.options.length < 2 || !q.options.every(text)) problems.push(`question ${i + 1} needs at least 2 options`);
    else if (!Number.isInteger(q.correctIndex) || q.correctIndex < 0 || q.correctIndex >= q.options.length) problems.push(`question ${i + 1} points outside its options`);
  });
  return problems;
}

export function viewPassageQuestions(c: PassageQuestionsContent): PassageQuestionsView {
  return { ...(c.instruction ? { instruction: c.instruction } : {}), passage: c.passage, questions: c.questions.map(({ question, options }) => ({ question, options })) };
}

export function parsePassageQuestionsAnswer(raw: unknown, c: PassageQuestionsContent): PassageQuestionsAnswer | null {
  const a = raw as { type?: unknown; selected?: unknown };
  if (a?.type !== 'passage_questions' || !Array.isArray(a.selected) || a.selected.length !== c.questions.length) return null;
```

```ts
export function validateMatching(content: unknown): string[] {

  const problems: string[] = [];
  if (!Array.isArray(c?.items) || c!.items.length === 0 || !c!.items.every(text)) problems.push('items must be a non-empty list of texts');
```

```ts
import type { LocalizedText } from '../i18n/localizedText';
import { gradeFillBlank } from '../tutoring/grading';

type Gap = { mode: 'typed'; answer: string; variants?: string[] } | { mode: 'select'; options: string[]; correctIndex: number } | { mode: 'bank'; answer: number };

export interface ClozeContent {
  instruction?: LocalizedText;
  text: string;
  gaps: Gap[];
  bank?: string[];
}
export interface ClozeView {
  instruction?: LocalizedText;
  text: string;
  gaps: ({ mode: 'typed' } | { mode: 'select'; options: string[] } | { mode: 'bank' })[];
  bank?: string[];
}
export interface ClozeAnswer {
  type: 'cloze';
  values: (string | number | null)[];
}

const GAP = /\{\{(\d+)\}\}/g;

export function validateCloze(content: unknown): string[] {
  const c = content as Partial<ClozeContent>;
  const problems: string[] = [];
  if (typeof c?.text !== 'string' || !c.text.trim()) return ['text must be a non-empty string'];
  const marks = [...c.text.matchAll(GAP)].map((m) => Number(m[1]));
  const gaps = Array.isArray(c.gaps) ? c.gaps : [];
  if (marks.length !== gaps.length) problems.push(`the text has ${marks.length} gaps but ${gaps.length} are defined`);
  else if (!marks.every((n, i) => n === i + 1)) problems.push('gaps must be numbered {{1}}, {{2}} … in order');
  gaps.forEach((g, i) => {
    if (g.mode === 'typed' && !(typeof g.answer === 'string' && g.answer.trim())) problems.push(`gap ${i + 1} needs an answer`);
    if (g.mode === 'select' && (!Array.isArray(g.options) || g.options.length < 2 || !Number.isInteger(g.correctIndex) || g.correctIndex < 0 || g.correctIndex >= g.options.length)) {
      problems.push(`gap ${i + 1} needs at least 2 options and a correct one`);
    }
    if (g.mode === 'bank') {
      if (!Array.isArray(c.bank) || c.bank.length === 0) problems.push(`gap ${i + 1} uses the bank but there is none`);
      else if (!Number.isInteger(g.answer) || g.answer < 0 || g.answer >= c.bank.length) problems.push(`gap ${i + 1} points outside the bank`);
    }
  });
  return problems;
}

export function viewCloze(c: ClozeContent): ClozeView {
  return {
    ...(c.instruction ? { instruction: c.instruction } : {}),
    text: c.text,
    gaps: c.gaps.map((g) => (g.mode === 'select' ? { mode: 'select', options: g.options } : { mode: g.mode })),
    ...(c.bank ? { bank: c.bank } : {}),
  };
}

export function parseClozeAnswer(raw: unknown, c: ClozeContent): ClozeAnswer | null {
  const a = raw as { type?: unknown; values?: unknown };
  if (a?.type !== 'cloze' || !Array.isArray(a.values) || a.values.length !== c.gaps.length) return null;
  const ok = a.values.every((v, i) => {
    if (v === null) return true;
    const g = c.gaps[i];
    if (g.mode === 'typed') return typeof v === 'string';
    const size = g.mode === 'select' ? g.options.length : (c.bank?.length ?? 0);
    return Number.isInteger(v) && (v as number) >= 0 && (v as number) < size;
  });
  return ok ? { type: 'cloze', values: a.values as (string | number | null)[] } : null;
}

// Each gap on its own: the same bank word may be right for two gaps (Review Focus 3).
export function gradeCloze(c: ClozeContent, a: ClozeAnswer): { itemResults: boolean[] } {
  return {
    itemResults: c.gaps.map((g, i) => {
      const v = a.values[i];
      if (v === null) return false;
      if (g.mode === 'typed') return gradeFillBlank({ textWithBlank: '___', correctAnswer: g.answer, acceptableVariants: g.variants }, String(v)) === 'correct';
      return g.mode === 'select' ? v === g.correctIndex : v === g.answer;
    }),
  };
}

export function correctTextCloze(c: ClozeContent): string[] {
  return c.gaps.map((g) => (g.mode === 'typed' ? g.answer : g.mode === 'select' ? g.options[g.correctIndex] : c.bank![g.answer]));
}
```

```ts
import type { ChatMessage } from '../providers/types';
import type { CefrLevel } from '../types';
import type { LocalizedText } from '../i18n/localizedText';
import { itemCount, isSetTaskType, validateTask, type SetTaskType, type TaskContent } from '../tasks/taskTypes';
import type { ExamPart } from './formats';

export interface TeilSetContent {
  taskType: SetTaskType;
  content: TaskContent;
  explanations: LocalizedText[];
}

export interface BundledSetFile {
  formatId: string;
  partId: string;
  version: string;
  sets: { id: string; set: TeilSetContent }[];
}

export function teilSetProblems(set: TeilSetContent, part: ExamPart): string[] {
  if (set.taskType !== part.taskType) return [`the part needs ${part.taskType}, not ${set.taskType}`];
  const problems = validateTask(set.taskType, set.content);
  if (problems.length > 0) return problems;
  const items = itemCount(set.taskType, set.content);
  if (items !== part.items) problems.push(`the part has ${part.items} items, the set has ${items}`);
  if (!Array.isArray(set.explanations) || set.explanations.length !== items) {
    problems.push(`${items} items need ${items} explanations, found ${set.explanations?.length ?? 0}`);
  } else if (set.explanations.some((e) => !e?.en?.trim() || !e?.de?.trim())) {
    problems.push('every explanation needs English and German');
  }
  return problems;
}

const SHAPE: Record<SetTaskType, string> = {
  passage_questions: '{"passage": "...", "questions": [{"question": "...", "options": ["...", "...", "..."], "correctIndex": 0}]}',
  matching: '{"items": ["..."], "targets": ["..."], "answers": [0], "allowNone": true | false}',
  cloze: '{"text": "... {{1}} ... {{2}} ...", "gaps": [{"mode": "select", "options": ["...", "...", "..."], "correctIndex": 0} | {"mode": "bank", "answer": 0}], "bank": ["..."] (only with bank gaps)}',
};

// Spec: pool sets are CEFR-style and German-only, in the part's exact format.
export function buildTeilSetPrompt(part: ExamPart, level: CefrLevel): { systemPrompt: string; messages: ChatMessage[] } {
  if (!isSetTaskType(part.taskType)) throw new Error(`No set generation for ${part.taskType}`);
  return {
    systemPrompt: [
      `You write one practice set for the exam part "${part.title.de}" at CEFR level ${level}.`,
      `Follow the real exam format. The task instructions are: ${part.instructions}`,
      `Write everything entirely in German, using vocabulary and grammar appropriate for CEFR level ${level}. The set must have exactly ${part.items} scored items.`,
      'For every item, add an explanation of the right answer in English and in simple German.',
      `Reply with only a JSON object: {"content": ${SHAPE[part.taskType]}, "explanations": [{"en": "...", "de": "..."}]}`,
    ].join('\n'),
    messages: [{ role: 'user', content: 'Write the set now.' }],
  };
}

export function parseTeilSetReply(text: string, part: ExamPart): TeilSetContent | null {
  const o = extractJsonObject(text);
  if (!o || !isSetTaskType(part.taskType)) return null;
  const set: TeilSetContent = { taskType: part.taskType, content: o.content as TaskContent, explanations: o.explanations as LocalizedText[] };
  return teilSetProblems(set, part).length === 0 ? set : null;
}
```
